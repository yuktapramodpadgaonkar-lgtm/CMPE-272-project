<?php
/**
 * Web cookie tracking for CMPE 272 lab:
 * - Last 5 previously visited product pages
 * - Visit counts per product (for "five most visited")
 *
 * Call track_product_page_visit($id) at the top of product.php BEFORE any HTML output.
 */

/** Cookie path = folder of current script (works in subdirectories on shared hosting). */
function sc_cookie_path(): string {
    $script = $_SERVER['SCRIPT_NAME'] ?? '/index.php';
    $dir = dirname(str_replace('\\', '/', $script));
    if ($dir === '/' || $dir === '.' || $dir === '') {
        return '/';
    }
    return rtrim($dir, '/') . '/';
}

/** Cookie lifetime: 30 days for recent list; 1 year for counts */
function sc_cookie_options(int $lifetimeSeconds): array {
    return [
        'expires'  => time() + $lifetimeSeconds,
        'path'     => sc_cookie_path(),
        'secure'   => !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off',
        'httponly' => false,
        'samesite' => 'Lax',
    ];
}

function sc_get_recent_product_ids(): array {
    if (empty($_COOKIE['sc_recent_products'])) {
        return [];
    }
    $decoded = json_decode($_COOKIE['sc_recent_products'], true);
    if (!is_array($decoded)) {
        return [];
    }
    $out = [];
    foreach ($decoded as $v) {
        $id = (int) $v;
        if ($id >= 1 && $id <= 10) {
            $out[] = $id;
        }
    }
    return $out;
}

/**
 * Maintain last 5 unique product IDs, most recent first.
 */
function sc_update_recent_products(int $productId): void {
    if ($productId < 1 || $productId > 10) {
        return;
    }
    $list = sc_get_recent_product_ids();
    $list = array_values(array_filter($list, static function ($id) use ($productId) {
        return (int) $id !== $productId;
    }));
    array_unshift($list, $productId);
    $list = array_slice($list, 0, 5);
    setcookie('sc_recent_products', json_encode($list), sc_cookie_options(60 * 60 * 24 * 30));
    $_COOKIE['sc_recent_products'] = json_encode($list);
}

function sc_get_visit_counts(): array {
    if (empty($_COOKIE['sc_product_visit_counts'])) {
        return [];
    }
    $decoded = json_decode($_COOKIE['sc_product_visit_counts'], true);
    if (!is_array($decoded)) {
        return [];
    }
    $counts = [];
    foreach ($decoded as $k => $v) {
        $id = (int) $k;
        if ($id >= 1 && $id <= 10) {
            $counts[$id] = max(0, (int) $v);
        }
    }
    return $counts;
}

/**
 * Increment visit count for a product (most visited tracking).
 */
function sc_increment_visit_count(int $productId): void {
    if ($productId < 1 || $productId > 10) {
        return;
    }
    $counts = sc_get_visit_counts();
    if (!isset($counts[$productId])) {
        $counts[$productId] = 0;
    }
    $counts[$productId]++;
    setcookie('sc_product_visit_counts', json_encode($counts), sc_cookie_options(60 * 60 * 24 * 365));
    $_COOKIE['sc_product_visit_counts'] = json_encode($counts);
}

/**
 * Top 5 product IDs by visit count (ties: lower product id first).
 */
function sc_get_top_five_most_visited_ids(): array {
    $counts = sc_get_visit_counts();
    if (empty($counts)) {
        return [];
    }
    $pairs = [];
    foreach ($counts as $id => $c) {
        $pairs[] = ['id' => (int) $id, 'c' => (int) $c];
    }
    usort($pairs, static function ($a, $b) {
        if ($b['c'] !== $a['c']) {
            return $b['c'] <=> $a['c'];
        }
        return $a['id'] <=> $b['id'];
    });
    $pairs = array_slice($pairs, 0, 5);
    return array_column($pairs, 'id');
}

function track_product_page_visit(int $productId): void {
    sc_update_recent_products($productId);
    sc_increment_visit_count($productId);
}
